\cvsubsection{Self-Evolving Agents}
\begin{cvitems}
  \item {\textbf{Cognitive Kernel: An Open-source Agent System towards Generalist Autopilots.} \textbf{Hongming Zhang}, Xiaoman Pan, Hongwei Wang, Kaixin Ma, Wenhao Yu, and Dong Yu. (NAACL 2025 Demo)\\
  \hspace*{1.5em}\textit{Introduces a general-purpose agent system for realizing self-evolving agents in real-world tasks.}}
  \item {\textbf{R-Zero: Self-Evolving Reasoning LLM from Zero Data.} Chengsong Huang, Wenhao Yu, Xiaoyang Wang, \textbf{Hongming Zhang}, Zongxia Li, Ruosen Li, Jiaxin Huang, Haitao Mi, and Dong Yu. (ICLR 2026)\\
  \hspace*{1.5em}\textit{Explores a novel approach to optimizing a model's task generation capability through self-evolution.}}
  \item {\textbf{WebEvolver: Enhancing Web Agent Self-Improvement with Co-evolving World Model.} Tianqing Fang, \textbf{Hongming Zhang}, Zhisong Zhang, Kaixin Ma, Wenhao Yu, Haitao Mi, and Dong Yu. (EMNLP 2025)\\
  \hspace*{1.5em}\textit{Applies self-evolving algorithms to the agentic domain through a co-evolving world model.}}
\end{cvitems}

\cvsubsection{Structured Reasoning Models}
\begin{cvitems}
  \item {\textbf{Parallel-R1: Towards Parallel Thinking via Reinforcement Learning.} Tong Zheng, \textbf{Hongming Zhang}, Wenhao Yu, Xiaoyang Wang, He Xing, Runpeng Dai, Rui Liu, Huiwen Bao, Chengsong Huang, Heng Huang, and Dong Yu. (ICLR 2026)\\
  \hspace*{1.5em}\textit{Introduces reinforcement learning for parallel thinking in structured reasoning models.}}
  \item {\textbf{Streaming Looking Ahead with Token-level Self-reward.} \textbf{Hongming Zhang}, Ruixin Hong, and Dong Yu. (Technical Report 2025)\\
  \hspace*{1.5em}\textit{Proposes token-level self-reward for efficient look-ahead reasoning in streaming settings.}}
  \item {\textbf{Scaling Test-Time Compute for Agentic Coding.} Joongwon Kim, Wannan Yang, Kelvin Niu, \textbf{Hongming Zhang}, Yun Zhu, Eryk Helenowski, Ruan Silva, Zhengxing Chen, Srinivasan Iyer, Manzil Zaheer, Daniel Fried, Hannaneh Hajishirzi, Sanjeev Arora, Gabriel Synnaeve, Ruslan Salakhutdinov, and Anirudh Goyal. (in submission to COLM 2026)\\
  \hspace*{1.5em}\textit{Studies how to scale test-time compute for structured reasoning in agentic coding tasks.}}
\end{cvitems}
